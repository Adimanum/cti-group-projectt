// ---------- H1a Step 1: successful sign-ins with a non-browser User-Agent (T1550.004) ----------
FROM tycoon2fa-hunt
| EVAL ua = TO_STRING(user_agent)
| WHERE log_type == "signin" AND result == "success" AND (ua LIKE "axios/*" OR ua LIKE "node-fetch*" OR ua LIKE "undici*" OR ua == "node" OR ua LIKE "python-requests*")
| STATS signins = COUNT(*), user_agents = VALUES(ua), ips = VALUES(src_ip) BY user

// ---------- H1  Step 2: non-browser sign-in -> new MFA method / device within 60 min (T1098.005) ----------
FROM tycoon2fa-hunt
| EVAL ua = TO_STRING(user_agent)
| WHERE (log_type == "signin" AND result == "success" AND (ua LIKE "axios/*" OR ua LIKE "node-fetch*" OR ua LIKE "undici*" OR ua == "node" OR ua LIKE "python-requests*"))
     OR (log_type == "audit" AND operation IN ("User registered security info", "Register device"))
| EVAL signin_ts = CASE(log_type == "signin", @timestamp), reg_ts = CASE(log_type == "audit", @timestamp)
| EVAL day = DATE_TRUNC(1 day, @timestamp)
| STATS odd_signin = MIN(signin_ts), registered = MAX(reg_ts), user_agents = VALUES(ua), ips = VALUES(src_ip) BY user, day
| WHERE odd_signin IS NOT NULL AND registered IS NOT NULL
| EVAL minutes_after = DATE_DIFF("minute", odd_signin, registered)
| WHERE minutes_after >= 0 AND minutes_after <= 60
// (ES|QL compares the first odd sign-in with the last registration of the same day; the exact
//  "followed within 60 min" logic is the EQL sequence at the end of this file)

// ---------- H2  Delete/hide inbox rule within 2 h after a sign-in from an unknown ASN (T1564.008) ----------
FROM tycoon2fa-hunt
| WHERE (log_type == "mailbox" AND operation == "New-InboxRule" AND (details LIKE "*DeleteMessage=True*" OR details LIKE "*MarkAsRead=True*"))
     OR (log_type == "signin" AND result == "success" AND TO_STRING(asn) NOT IN ("64500", "64501"))
| EVAL rule_ts = CASE(log_type == "mailbox", @timestamp), risky_ts = CASE(log_type == "signin", @timestamp)
| EVAL day = DATE_TRUNC(1 day, @timestamp)
| STATS risky_signin = MIN(risky_ts), rule_created = MAX(rule_ts), asn_names = VALUES(asn_name), rule = VALUES(details) BY user, day
| WHERE risky_signin IS NOT NULL AND rule_created IS NOT NULL
| EVAL minutes_after = DATE_DIFF("minute", risky_signin, rule_created)
| WHERE minutes_after >= 0 AND minutes_after <= 120

// ---------- H3a Graph burst, raw: >= 20 calls in one session within 60 s (T1087.004) ----------
FROM tycoon2fa-hunt
| WHERE log_type == "graph"
| STATS calls = COUNT(*), endpoints = COUNT_DISTINCT(request_uri), first_call = MIN(@timestamp), last_call = MAX(@timestamp) BY session_id, user, src_ip, app_name
| EVAL seconds = DATE_DIFF("second", first_call, last_call)
| WHERE calls >= 20 AND seconds <= 60
| SORT first_call

// ---------- H3  Graph burst, tuned: + at least 8 different endpoints ----------
FROM tycoon2fa-hunt
| WHERE log_type == "graph"
| STATS calls = COUNT(*), endpoints = COUNT_DISTINCT(request_uri), first_call = MIN(@timestamp), last_call = MAX(@timestamp) BY session_id, user, src_ip, app_name
| EVAL seconds = DATE_DIFF("second", first_call, last_call)
| WHERE calls >= 20 AND seconds <= 60 AND endpoints >= 8
| SORT first_call
